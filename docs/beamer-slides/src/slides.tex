\begin{frame}{What is the HHL algorithm?}

    \iconcite{Harrow_2009} \\
    Let $Ax = b$ a linear system, $A\in\mathbb{R}^{N_b \times N_b}, b \in \mathbb{R}^{N_b}$
    \vspace{\baselineskip}
    \begin{itemize}
        \item \textbf{Input limitations:} $A$ sparse and hermitian
        \item \textbf{Output limitations:}
            HHL \textbf{does not} actually compute $x$ (\textit{amplitude encoding})
            \begin{equation*}
                \ket{x} = \sum_{j=0}^{N_b-1} x_j \ket{j}
            \end{equation*}
        \item \textbf{Logarithmic scaling in $N_b$:} \\ 
        \begin{equation*}
            \mathcal{O}(\log N_b \kappa^2/\epsilon) \quad \text{vs} \quad \mathcal{O}(N_b \kappa)
        \end{equation*}
        $\kappa$: condition number of $A$, $\epsilon$: desired accuracy
    \end{itemize}
\end{frame}

\begin{frame}{Basic idea}
    Given the eigenvalue decomposition (EVD) of the matrix, $A = \sum_i \lambda_i{u}_i{u}_i^\top$, its inverse can be computed as
    \begin{equation*}
        A^{-1} = \sum_i \lambda_i^{-1}{u}_i{u}_i^\top
    \end{equation*}
    Thus, if we rewrite ${b}$ in the eigenvector basis, ${b} = \sum_i \beta_i {u}_i$, ${x}$ will be
    \begin{align*}
        {x} 
        = A^{-1}{b}
        = \sum_i \lambda_i^{-1}{u}_i{u}_i^\top {b}
        = \sum_i \lambda_i^{-1} \beta_i {u}_i
    \end{align*}
\end{frame}

\begin{frame}{HHL implementation}
    \only<1>{
        Registers:
        \begin{itemize}
        \item \textit{b-register}: components of ${b}, {x}$, (\textit{amplitude encoding}), $n_b$ qubits, with $N_b = 2^{n_b}$
        \item \textit{c-register}: phase of the eigenvalues of $A$, $n_c$ qubits, $N_c = 2^{n_c}$
        \item \textit{ancilla}: inversion and measurement
    \end{itemize}
    }
    \only<2>{
        Initial state:
        \begin{equation*}
            \ket{\Psi_0} = \ket{0}^{\otimes n_b}\ket{0}^{\otimes n_c}\ket{0}
        \end{equation*}   
    }
    \only<3>{
        Steps:
        \begin{multicols}{2}
        \begin{enumerate}
            \item State preparation
            \item QPE
            \item Ancilla qubit rotation
            \item Inverse QPE
        \end{enumerate}
        \end{multicols}
    }

\begin{frame}{State preparation}
    \textbf{State preparation:} rotate the qubits in the b-register to have amplitudes corresponding to the coefficients of ${b}$ (\textit{amplitude encoding})

    \vspace{\baselineskip}
    A simple example: ${b} = (0,1), N_b = 2 \implies n_b = 1$

    \begin{center}
    \begin{quantikz}
        % Input/Output register (multi-qubit bundle)
        \lstick{b-reg} &
            \qwbundle{1} &
            \gate[style={fill=green!20}]{X} &
            \rstick{$\ket{1}$}
    \end{quantikz}
    \end{center}

    \vspace{\baselineskip}
    Recall:
    \begin{itemize}
        \item state preparation step specific to each ${b}$ 
        \item ${b}$ needs to be normalized: $\sum_i |\tilde{b}_i|^2 = 1$
        \item output: $\ket{\Psi_0} = \ket{{b}}\ket{0}^{\otimes n_c}\ket{0}$
    \end{itemize}
\end{frame}

\begin{frame}{Quantum Phase Estimatation (QPE)}
    \textbf{Goal:} estimate the phase of the eigenvalues of $A$ \\
    \vspace{\baselineskip}
    Issues:
    \begin{itemize}
        \item How to encode matrix $A$ in the circuit? \textit{Hamiltonian encoding}
        \item How to estimate its eigenvalues? \textit{Quantum Phase Estimation}
    \end{itemize}
\end{frame}

\begin{frame}{Hamiltonian encoding}
    In general, $A$ non-unitary: we'll use
    \begin{equation*}
        U = e^{iAt}
    \end{equation*}
    which is unitary and can be approximated by solving a \textit{Hamiltonian simulation} problem
    
    \begin{center}
    \begin{quantikz}&
            \qwbundle{n_b} &
            \gate[style={fill=green!20}]{e^{iAt}} &
    \end{quantikz}
    \end{center}

    \begin{problem}[Hamiltonian Simulation Problem]
        Given a  Hamiltonian ($2^{n_b}\times 2^{n_b}$ Hermitian matrix), $A$, a time $t$ and a tolerance $\epsilon$, find $U$ such that 
        \begin{equation*}
            ||U-e^{iAt}|| \leq \epsilon
        \end{equation*}
    \end{problem}
    Algorithm: Trotterization + Pauli gadget ($A = \sum_j A_j$, $U \approx \prod_j e^{iA_jt}$)
\end{frame}

\begin{frame}{Quantum Phase Estimation (QPE)}
    Eigenvalue connection: $\text{eig}(U) = e^{i\lambda_j t}$, where $\lambda_j$ eigenvalues of $A$ \\

        \item Controlled rotations

\begin{frame}{QPE: Superpostion}
    First step: superimpose the qubits:
    \begin{equation*}
        \ket{\Psi_2} = \ket{b} \frac{1}{2^{\frac{n_c}{2}}} (\ket{0}+\ket{1})^{\otimes n} \ket{0}_a
    \end{equation*}
    This is a prerequisite to exploit \textit{phase kickback} in subsequent steps
\end{frame}

\begin{frame}[allowframebreaks]{QPE: Controlled Rotations}

    For the moment, let's assume $\ket{b}$ eigenvector of $U$ with eigenvalue $e^{i2\pi \phi}$:
    \begin{equation*}
        U \ket{b} = e^{i2\pi \phi} \ket{b}
    \end{equation*}

    Recall that we can always express $\ket{b}$ in terms of the basis of eigenvectors of $U$
    \begin{equation*}
        \ket{b} = \sum_j \beta_j \ket{u_j}
    \end{equation*}

    \framebreak
    
    If the control is $0$, $\ket{b}$ is not affected, otherwise $U$ is applied. This is equivalent to (\textit{phase kickback}):
    \begin{align*}
        \ket{\Psi_3} 
        &= (\frac{1}{2^{\frac{n_c}{2}}} (\ket{b}\ket{0}+U^{2^{n_c}-1}\ket{b}\ket{1}) \otimes \cdots \otimes (\ket{0}+U^{2^0}\ket{b}\ket{1})) \otimes \ket{0}_a \\
        &= \ket{b} \otimes (\frac{1}{2^{\frac{n_c}{2}}} (\ket{0}+e^{i2\pi \phi (2^{n_c}-1)}\ket{1}) \otimes \cdots \otimes (\ket{0}+e^{i2\pi \phi 2^{0}}\ket{1})) \otimes \ket{0}_a \\
        %&= \ket{b} \otimes (\frac{1}{2^{\frac{n_c}{2}}}\bigotimes_{j=1}^{n_c}(\ket{0}+e^{2\pi i \phi 2^{n_c-j}}\ket{1})) \otimes \ket{0}_a \\
        &= \ket{b} \otimes (\frac{1}{\sqrt{N_c}}\sum_{k=1}^{N_c-1} e^{i2\pi \phi k} \ket{k}) \otimes \ket{0}_a 
    \end{align*}
    \textbf{Motivation:} transfer phase information from $U$ to the \textit{c-registers}!
\end{frame}

\begin{frame}{Quantum Fourier Transform}
    QFT maps a \textbf{state in the computational basis to a state in the Fourier basis}. Let $\ket{x} = \sum_{j=0}^{N_c} x_j \ket{j}$ expressed in the computational basis
    \begin{equation*}
        \ket{j} \to U_{\text{QFT}}\ket{j} = \frac{1}{\sqrt{N_c}}\sum_{k=0}^{N_c} e^{i2\pi jk /N_c} \ket{k}
    \end{equation*}
    leading to $\ket{x}=\sum_{j=0}^{N_c} x_j \ket{j} = \sum_j x_j QFT(\ket{j})$ expressed in the fourier basis. Unitary matrix representation:
    \begin{equation*}
        U_{\text{QFT}} = \frac{1}{\sqrt{N_c}} 
        \begin{bmatrix}
            1 & \ldots & 1 \\
            1 & \ldots & e^{i2\pi (N_c-1)/ N_c} \\
            \vdots & \ddots & \vdots \\
            1 & \ldots & e^{i2\pi (N_c-1)(N_c-1)/ N_c}
        \end{bmatrix}
    \end{equation*}
\end{frame}

\begin{frame}{QPE: Inverse Quantum Fourier Transform}
    Observe that
    \begin{equation*}
        \ket{\Psi_3} = \ket{b} \otimes (\frac{1}{\sqrt{N_c}}\sum_{k=0}^{N_c-1} e^{i2\pi \phi k } \ket{k}) \otimes \ket{0}_a 
    \end{equation*}
    stores phase information ($\ket{N_c\phi}$) in the Fourier basis. Indeed,
    \begin{equation*}
        U_{\text{QFT}}\ket{N_c\phi} = \frac{1}{\sqrt{N_c}}\sum_{k=0}^{N_c-1} e^{i2\pi N_c\phi k / N_c} \ket{k}
    \end{equation*}
    Thus, $\text{IQFT}(\text{QFT}(\ket{N\phi})) = \ket{N\phi}$
    
\end{frame}

\begin{frame}{QPE: Inverse Quantum Fourier Transform}
    To extract the phase information from the \textit{c-registers} of $\ket{\Psi_3}$, IQFT is applied:
    \begin{align*}
        \mathrm{IQFT}(\frac{1}{\sqrt{N_c}}\sum_{k=0}^{N_c-1}(e^{i2\pi \phi k}\ket{k})) &=  \frac{1}{\sqrt{N_c}}\sum_{k=0}^{N_c-1}(e^{i2\pi \phi k}\mathrm{IQFT}(\ket{k})) \\
        &= \frac{1}{\sqrt{N_c}}\sum_{k=0}^{2^{n_c-1}}(e^{i2\pi \phi k}(\frac{1}{\sqrt{N_c}}\sum_{j=0}^{N_c-1}e^{ -i2\pi jk/N_c}\ket{j})) \\
        &= \frac{1}{N_c}\sum_{k=0}^{N_c-1}(\sum_{j=0}^{N_c-1}e^{i2\pi k(\phi - j/N_c)}\ket{j})
    \end{align*}
\end{frame}

\begin{frame}{Ancilla qubit}

    \begin{itemize}
        \item if $\ket{0}$: throw away the results
        \item if $\ket{1}$: store the results
    \end{itemize}

        \ket{\Psi_6} = \frac{1}{\sum_{j=0}^{2^{n_b}-1} |\frac{\beta_j N_c}{\tilde{\lambda}_j}|^2}\sum_{j=0}^{2^{n_b}-1} \beta_j\ket{u_j}\ket{\tilde{\lambda_j}}\frac{C}{\tilde{\lambda}_j}\ket{1}_a
    \end{equation*}

\begin{frame}{Hyperparameters selection}
    Three hyperparameters: $(n_c,t,C)$. To set them:\\
    \vspace{\baselineskip}
    \begin{itemize}
        \item $(t,n_c)$ determine a discrete grid to estimate the eigenvalues of $A$, $\lambda_j$
        \item $\ket{0\dots 0}$ cannot store any $\lambda_j$
        \item $C$ has to be lower than $\min_j \lambda_j$ but as high as possible
    \end{itemize}
\end{frame}

\begin{frame}{Hyperparameters selection}
    Recall $\phi = \frac{\lambda_j t}{2\pi}$ with $\phi \in [0,1)$ and that the \textit{c-registers} store $\ket{N_c\phi}$. Then \\
    \vspace{\baselineskip}
    \begin{itemize}
        \item \textbf{Scale:} 
        \begin{equation*}
            \phi < 1 \implies t < \frac{2\pi}{\max_j \lambda_j}
        \end{equation*}

        \item \textbf{Sensibility:}
        \begin{equation*}
            \Delta \phi = \frac{1}{N_c} \implies \Delta \lambda = \frac{2\pi}{tN_c}
        \end{equation*}
    \end{itemize}
    Moreover, the fact that $\ket{0\dots 0}_c$ cannot store any $\tilde{\lambda}_j$ leads to $t > \frac{2\pi}{N_c\min_j\lambda_j}$ and $N_c > \frac{\max_j \lambda_j}{\min_j \lambda_j}=\kappa$
\end{frame}


\begin{frame}{Inverse QPE}
    \begin{equation*}
        \ket{\Psi_6} = \frac{1}{\sum_{j=0}^{2^{n_b}-1} |\frac{\beta_j N_c}{\tilde{\lambda}_j}|^2}\sum_{j=0}^{2^{n_b}-1} C\frac{\beta_j}{\tilde{\lambda}_j}\ket{u_j}\ket{\tilde{\lambda_j}}\ket{1}_a
    \end{equation*}

    \textbf{Issue:} \textit{b-registers} storing the solution in the desired form are entangled with the \textit{c-registers}
    
\end{frame}

\begin{frame}{Inverse QPE: Quantum Fourier Transform}
    Firstly, QFT is applied to the \textit{c-register} qubits:

    \begin{equation*}
        \mathrm{QFT}(\ket{\tilde{\lambda_j}}) = \frac{1}{2^{\frac{n}{2}}}\sum_{k=0}^{N_c-1} e^{2\pi ik\tilde{\lambda}_j/N_c}\ket{k}
    \end{equation*}
    leading to 
    \begin{align*}
        \ket{\Psi_7} = \frac{1}{\sum_{j=0}^{2^{n_b}-1} |\frac{\beta_j N_c}{\tilde{\lambda}_j}|^2} \sum_{j=0}^{2^{n_b}-1} C\frac{\beta_j}{\tilde{\lambda}_j}\ket{u_j}(\frac{1}{2^{\frac{n}{2}}}\sum_{k=0}^{N_c-1} e^{2\pi ik\tilde{\lambda}_j/N_c}\ket{k})\ket{1}_a
    \end{align*}
\end{frame}

\begin{frame}{Inverse QPE: Controlled rotations}
    Then controlled rotations with $U^{-1}$ are applied as before

    \begin{align*}
        &\sum_{j=0}^{2^{n_b}-1} C\frac{\beta_j}{\tilde{\lambda}_j}\ket{u_j}(\frac{1}{2^{\frac{n}{2}}}\sum_{k=0}^{N_c-1} e^{-i\lambda_j tk}e^{2\pi ik\tilde{\lambda}_j/N_c}\ket{k}) \\
        &= \sum_{j=0}^{2^{n_b}-1} C\frac{\beta_j}{\tilde{\lambda}_j}\ket{u_j}\sum_{k=0}^{N_c-1} \ket{k} 
        = C\ket{x}\sum_{k=0}^{N_c-1} \frac{1}{2^{\frac{n}{2}}}\ket{k}
    \end{align*}

    leading to 

    \begin{align*}
        \ket{\Psi_8} = \frac{C}{2^{\frac{n}{2}}\sum_{j=0}^{2^{n_b}-1} |\frac{\beta_j N_c}{\tilde{\lambda}_j}|^2} \ket{x}\sum_{k=0}^{N_c-1} \frac{1}{2^{\frac{n}{2}}}\ket{k}\ket{1}_a
    \end{align*}
\end{frame}

\begin{frame}{Inverse QPE}
    The clock-qubits and \textit{b-registers} are now unentagled. By applying the Hadamard gates to the clock qubits we have
    \begin{align*}
        \ket{\Psi_9} &= \frac{C}{\sum_{j=0}^{2^{n_b}-1} |\frac{\beta_j N_c}{\tilde{\lambda}_j}|^2} \ket{x}_b \ket{0}_c^{\otimes n}\ket{1}_a
    \end{align*}
    where the solution $\ket{x}$ is stored in the \textit{b-registers} successfully.
\end{frame}

\begin{frame}{Conclusions}
    Pros:
    \begin{itemize}
        \item A quantum algorithm for linear systems
        \item Instructive: several subroutines common to many other quantum algorithms (QFT, QPE, Hamiltonian simulation)
    \end{itemize}

    Cons:
    \begin{itemize}
        \item Limited: $x^\top M x$ 
        \item Deep circuit: noise on real quantum hardware
    \end{itemize}
\end{frame}
